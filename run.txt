python main.py --mode train

# Train with custom parameters
python main.py --mode train --block_size 64 --embed_dim 128 --epochs 10 --batch_size 32

Run in eval mode:-
python main.py --mode evaluate

Run compression analysis:-
python main.py --mode analyze

Visualization
import matplotlib.pyplot as plt

def plot_training_results(train_losses, val_accuracies, block_size):
    plt.figure(figsize=(12, 5))
    
    plt.subplot(1, 2, 1)
    plt.plot(train_losses)
    plt.title('Training Loss')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    
    plt.subplot(1, 2, 2)
    plt.plot(val_accuracies)
    plt.title('Validation Accuracy')
    plt.xlabel('Epoch')
    plt.ylabel('Accuracy (%)')
    
    plt.suptitle(f'BCM Transformer with Block Size {block_size}')
    plt.tight_layout()
    plt.savefig(f'results_block_size_{block_size}.png')
    plt.show()
